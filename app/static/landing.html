Get running in seconds

Point your Anthropic SDK at the proxy — that's it.

# 1. Start the proxy
uvicorn app.main:app --host 0.0.0.0 --port 4000

# 2. Issue a virtual key from the admin dashboard (/ui/admin)
#    then point the Anthropic SDK at the proxy
export ANTHROPIC_BASE_URL="http://localhost:4000"
export ANTHROPIC_API_KEY="ccm_your_issued_key"

python -c "import anthropic; print(anthropic.Anthropic().messages.create(
  model='claude-sonnet-4-6',
  max_tokens=64,
  messages=[{'role':'user','content':'hi'}]))"
